"""Development-only check of context formulation after identifying int8 degradation."""
import json
from pathlib import Path
from entity_model import EntityModel,task_schema
ROOT=Path(__file__).resolve().parents[1]
rows=[r for r in json.loads((ROOT/'data/entity_evaluation.json').read_text(encoding='utf-8')) if r['split']=='development']
model=EntityModel(quantize=False)
out=[]
for mode in ('plain_200','plain_100','binary_100'):
 for person in ('lenin','stalin'):
  subset=[r for r in rows if r['person']==person]
  texts=[]
  for r in subset:
   pos=r['context_offset']; w=100 if mode.endswith('100') else 200
   texts.append(r['context'][max(0,pos-w):pos+len(r['matched'])+w])
  if mode.startswith('binary'):
   name={'lenin':'Владимир Ленин','stalin':'Иосиф Сталин'}[person]
   schema={'mention':{'labels':{'historical_person':f'{name} как человек, его биография, слова, политика или действия.','not_historical_person':'Название места, учреждения, корабля, памятника, награды или другой человек.'},'multi_label':True,'class_act':'softmax','cls_threshold':0.0}}
   task='mention'
  else: schema=task_schema(person);task='meaning'
  with model.torch.inference_mode(): preds=model.model.batch_classify_text(texts,schema,batch_size=4,include_confidence=True,max_len=768)
  for r,p in zip(subset,preds):
   pred=max(p[task],key=lambda v:v['confidence'])
   ref=r['reference_category']
   if mode.startswith('binary') and ref!='historical_person': ref='not_historical_person'
   out.append(dict(mode=mode,evaluation_id=r['evaluation_id'],reference=ref,prediction=pred))
 for_mode=[v for v in out if v['mode']==mode]
 print(mode,sum(v['reference']==v['prediction']['label'] for v in for_mode),'/',len(for_mode),flush=True)
 for v in for_mode: print(v['evaluation_id'],v['reference'],v['prediction'],flush=True)
(ROOT/'output/data_cleaning/neural/development_context_probe.json').write_text(json.dumps(out,ensure_ascii=False,indent=2),encoding='utf-8')
