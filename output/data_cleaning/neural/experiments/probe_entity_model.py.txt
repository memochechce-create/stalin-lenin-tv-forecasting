"""Development-only diagnostic: full precision, task descriptions and target markers."""
import json
import time
from pathlib import Path
from entity_model import EntityModel,task_schema,model_text

ROOT=Path(__file__).resolve().parents[1]
rows=json.loads((ROOT/'data/entity_evaluation.json').read_text(encoding='utf-8'))
rows=[r for r in rows if r['split']=='development']
model=EntityModel(quantize=False)
tests=[('sentiment_en','I love this wonderful product.',{'sentiment':['positive','negative']}),
       ('sentiment_ru','Мне очень понравился этот прекрасный фильм.',{'sentiment':['positive','negative']}),
       ('simple_street','На площади Ленина произошла авария.',{'meaning':['person','street or square','ship','other person']}),
       ('simple_person','Владимир Ленин выступил с речью перед рабочими.',{'meaning':['person','street or square','ship','other person']})]
for name,text,schema in tests:
 with model.torch.inference_mode(): print(name,model.model.classify_text(text,schema,include_confidence=True),flush=True)
results=[]
for mode in ('descriptions_ru','labels_ru','labels_en'):
 for person in ('lenin','stalin'):
  subset=[r for r in rows if r['person']==person][:6]
  if mode=='descriptions_ru': schema=task_schema(person)
  elif mode=='labels_ru': schema={'meaning':['исторический человек','название места или объекта','памятник или портрет','другой человек','ошибка распознавания речи','неясно']}
  else: schema={'meaning':['historical person','named location or object','monument or portrait','other person','transcription error','unclear']}
  with model.torch.inference_mode(): preds=model.model.batch_classify_text([model_text(r) for r in subset],schema,batch_size=4,include_confidence=True,max_len=768)
  for r,p in zip(subset,preds):
   print(mode,r['evaluation_id'],r['reference_category'],p,flush=True)
   results.append(dict(mode=mode,evaluation_id=r['evaluation_id'],reference=r['reference_category'],prediction=p))
(ROOT/'output/data_cleaning/neural/development_probe.json').write_text(json.dumps(results,ensure_ascii=False,indent=2),encoding='utf-8')
